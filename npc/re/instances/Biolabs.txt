// 3rd arg = 1 to block, 0 to disable blocks
function	script	F_SetBlocker	{
	.@setValue = getarg(2);
	.@blockArraySize = getarraysize(getarg(0));
	.@cellArraySize = getarraysize(getarg(1));

	for(.@i = 0; .@i < .@blockArraySize; .@i++) {
		if (.@setValue)
			enablenpc instance_npcname("#block_" + getelementofarray(getarg(0), .@i));
