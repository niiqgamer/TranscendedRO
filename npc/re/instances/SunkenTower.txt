		mes "You don't want to do anything with me, go away!";

	}

	mes "I can ^FF0000RESET^000000 Sunken Tower Instance for you.";
	mes "Do you interest?";
	next;
	switch(select("Please wait.:Yes, I'm in.")) {
	case 1:
		mes "All right, don't forget we don't have much time.";
		end;
	case 2:
		.@itemNeeded = 6417;
		.@itemCount = 3;
		if (countitem(.@itemNeeded) < .@itemCount) {
			mes "["+strnpcinfo(1)+"]", "What gives? You don't have enough " + mesitemlink(.@itemNeeded) + ".";
			end;
		}
		if (checkquest(12541,PLAYTIME) != -1){
			erasequest 12541;
		}
		delitem .@itemNeeded, .@itemCount;
		specialeffect2 300;
		dispbottom "The Sunken Tower's distortion is now RESET!!!";
		mes "Successful Reset Instance!";
		close3;
		end;
	}

OnInit:
	initnpctimer;
	end;

OnTimer1000:
    showscript(" ! Reset Instance ! ");
    setnpctimer 0;
    end;
}

alberta,214,74,5	script	Leticia#s_alberta	4_4JOB_LETICIA,{
	cutin "4job_leticia_01.png",2;
	if (BaseLevel < 40) {
		mes "["+strnpcinfo(1)+"]", "You're much too weak! Come back when you're stronger.";
		close3;
	}

	// ID of cooldown quest
	.@cooldown = 12541; 

	if(checkquest(.@cooldown,PLAYTIME) == 2){
		erasequest .@cooldown;
	}
	else if (checkquest(.@cooldown,PLAYTIME) != -1){
		mes "["+strnpcinfo(1)+"]", "The Sunken Tower's distortion is still affecting you.", "Come back when it's worn off.";
		close3;
	}

	.@party_id = getcharid(1);
	.@instance$ = "Sunken Tower";

	if (.@party_id < 1) {
		mes "["+strnpcinfo(1)+"]", "Hmm? Where's your party? Please form a party even if you're going alone.";
		close3;
	}

	if (BaseLevel >= 260) {
		.@i = 260;
		.@maxLvl = 275;
	}
	else {
		for (.@i = 40; .@i < 275; .@i+=20) {
			if (BaseLevel >= .@i && BaseLevel < .@i + 20) {
				.@maxLvl = .@i + 19;
				break;
			}
		}
	}
	
	mes "["+strnpcinfo(1)+"]", "Wanna help clean up the Sunken Tower?";
	next;
	switch(select("Cancel", "Create " + .@instance$ + " " + .@i + "~" + .@maxLvl,"Enter " + .@instance$ + " " + .@i + "~" + .@maxLvl)){
		case 2:
			if(is_party_leader() == false){
				mes "You're not the leader of this party.";
				close3;
			}
			if(!instance_check_party(.@party_id, 1, .@i, .@maxLvl)) {
				mes "["+strnpcinfo(1)+"]", "Looks like someone in your party isn't eligible to enter that tier.";
				close3;
			}
			if (instance_create(.@instance$) < 0) {
				mes "["+strnpcinfo(1)+"]", "Error creating instance.";
				close3;
			}
			setinstancevar('rank, (.@i - 40) / 20, instance_id(IM_PARTY));
			close3;
		case 3:
			if (instance_id(IM_PARTY) > 0) {
				if (getinstancevar('partyCount, instance_id(IM_PARTY)) > 0) {
					mes "["+strnpcinfo(1)+"]", "The clean up is already in progress. I can't let you join anymore.";
					close3;
				}
			}
			switch(instance_enter(.@instance$)){
				case IE_OK:
					if (isbegin_quest(.@cooldown) == 0) {
						setquest .@cooldown;
						mapannounce "alberta", "" + strcharinfo(0) + " of the party, " + getpartyname( getcharid(1) ) + ", is entering " + .@instance$ + ".",bc_map,"0x00FF99";
					}
					close3;
				case IE_NOMEMBER:
				case IE_OTHER:
				case IE_NOINSTANCE:
				default:
					if (is_party_leader(.@party_id) == false) {
						mes "^ff0000Please note that this Memorial Dungeon cannot be accessed for 72 hours after your last visit.^000000";
						next;
					}
					mes "^ff0000Your party leader has not yet created the dungeon.^000000";
					close3;
			 }
		case 1:
			close3;
	}
	close3;

OnInit:
	initnpctimer;
	end;

OnTimer1000:
    showscript(" ! Sunken Tower ! ");
    setnpctimer 0;
    end;
}

1@ch_u,8,55,5	script	Leticia#s_tower	4_4JOB_LETICIA,{
	cutin "4job_leticia_01.png",2;
	if (!is_party_leader()) {
		mes "["+strnpcinfo(1)+"]", "Who are you? Don't talk to me unless you're the leader.";
		close3;
	}
	'partyId = getcharid(1);

	L_MainMenu:
		if (!'hasTalked) {
			mes "["+strnpcinfo(1)+"]", "Looks like the water is still flowing in.", "I'd say there's about 30 minutes until you're all underwater.", "Better hurry up.";
		} else {
			mes "["+strnpcinfo(1)+"]", "Time's running out now. What are you gonna do?";
		}
		mes "---------------";
		mes "Current Difficulty: ^0000CCLevel " + 'difficulty;
		'hasTalked = 1;
		next;
		switch(select("Hold on a moment!", "Let's go.", "^0000CCChange Difficulty")) {
			case 1:
				close3;
			case 2:
				donpcevent instance_npcname("#s_tower_controller")+"::OnFirstStart";
				hideonnpc instance_npcname(strnpcinfo(0));
				close3;
			case 3:
				// You can change the item needed and required amount here
				.@itemNeeded = 1000372;
				.@itemCount = 10;
				mes "["+strnpcinfo(1)+"]", "With 10 " + mesitemlink(.@itemNeeded) + ", you can increase the number and strength of monsters.", "It goes without saying that you'll get more experience too.";
				mes "---------------";
				mes "Current Difficulty: ^0000CCLevel " + 'difficulty;
				next;
				.@selection = select("Nevermind.", "Difficulty 1", "Difficulty 2", "Difficulty 3", "Difficulty 4", "Difficulty 5");
				if (.@selection == 1)
					close3;

				if (countitem(.@itemNeeded) < .@itemCount) {
					mes "["+strnpcinfo(1)+"]", "What gives? You don't have enough " + mesitemlink(.@itemNeeded) + ".";
					close3;
				}

				if ('level == (.@selection - 1)) {
					mes "["+strnpcinfo(1)+"]", "What are you doing? The current difficulty is already ^0000CCLevel " + 'difficulty + ".";
					close3;
				}

				mes "["+strnpcinfo(1)+"]", "You want to change the difficulty to ^0000CCLevel " + (.@selection - 1) + "?";
				next;
				if (select("Nevermind.", "Yeah, change it.") == 1) {
					close3;
				}

				'difficulty = .@selection - 1;
				mapannounce 'map$, "Leticia: The difficulty is now level " + 'difficulty + ".", bc_map,0xCC6666,FW_NORMAL,14;
				delitem .@itemNeeded, .@itemCount;
				close3;
		}

	close3;
	end;
}
