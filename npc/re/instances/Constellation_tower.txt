1@ch_t,1,1,0	script	#ctower_control	-1,{
    end;

OnInstanceInit:
    instance_enable("#oscar_ctower_1",true);
    instance_event(strnpcinfo(0),"OnCreateFloor",false);
	for ( .@i = 1; .@i <= 5; ++.@i ) {
		instance_enable("#ctower_warp_" + .@i, true);
	}
end;

OnCreateFloor:
	freeloop(true);
    for ( .@i = 1; .@i <= 75; ++.@i ){
        .@mvp_id = 0;
        .@mob_id = 0;
        .@type = rand(2,6);
        .@amount = rand(.@type,.@type + 8);
		deletearray .@mob_list[0];
		deletearray .@floor$[0];
        for ( .@x = 0; .@x < .@type; ++.@x ) {
            .@mob_id = rand(20998,21046);
            while(inarray(.@mob_list,.@mob_id) != -1)
               .@mob_id = rand(20998,21046);
            .@mob_list[.@x] = .@mob_id; 
        }
        for ( .@x = 0; .@x < .@amount; ++.@x ){
            if(.@x <= (.@type - 1))
                .@floor$[.@x] = .@mob_list[.@x];
            else
                .@floor$[.@x] = .@mob_list[rand(getarraysize(.@mob_list))];
        }
        if ( .@i % 5 == 0 ) {
            .@mvp_id = rand(21325,21358);
            while( inarray(.@mvp_list,.@mvp_id) != -1 )
                .@mvp_id = rand(21325,21358);
            .@mvp_list[getarraysize(.@mvp_list)] = .@mvp_id;
			setinstancevar("floor_" + .@i + "_mvp",.@mvp_id);
        }
        setinstancevar("floor_" + .@i + "$",implode(.@floor$,":"));
    }
	freeloop(false);
end;
}

//= Calculate Stats
function	script	ctower_calc_stats	{
	.@gid = getarg(0);
	if( unitexists(.@gid) ){   
		getunitdata .@gid,.@data;
		.@HP = getinstancevar("HP");
		.@ATK = getinstancevar("ATK");
		.@RES = getinstancevar("RES");
		if( .@HP ){
			.@MOB_HP = min(2000000000, .@data[UMOB_MAXHP] + (300000000 * .@HP));
			setunitdata .@gid,UMOB_MAXHP,.@MOB_HP;
			setunitdata .@gid,UMOB_HP,.@MOB_HP;
			setinstancevar("BOSS_HP", .@MOB_HP);
		}
		if( .@ATK ){
			
			.@MOB_ATK = min(65000, .@data[UMOB_ATKMIN] + (5000 * .@ATK));
			.@MOB_ATK2 = min(65000, .@data[UMOB_ATKMAX] + (5000 * .@ATK));
			
			.@MOB_MATK = min(65000, .@data[UMOB_MATKMIN] + (5000 * .@ATK));
			.@MOB_MATK2 = min(65000, .@data[UMOB_MATKMAX] + (5000 * .@ATK));
			
			.@STR = .@data[UMOB_STR] + (60 * .@ATK);
			.@AGI = .@data[UMOB_AGI] + (60 * .@ATK);
			.@INT = .@data[UMOB_INT] + (60 * .@ATK);
			.@VIT = .@data[UMOB_VIT] + (60 * .@ATK);
			.@DEX = .@data[UMOB_DEX] + (60 * .@ATK);
			.@LUK = .@data[UMOB_LUK] + (60 * .@ATK);
			
			setunitdata .@gid, UMOB_ATKMIN, .@MOB_ATK;
			setunitdata .@gid, UMOB_ATKMAX, .@MOB_ATK2;
			setunitdata .@gid, UMOB_MATKMIN, .@MOB_MATK;
			setunitdata .@gid, UMOB_MATKMAX, .@MOB_MATK2;
			setunitdata .@gid, UMOB_STR, .@STR;
			setunitdata .@gid, UMOB_AGI, .@AGI;
			setunitdata .@gid, UMOB_VIT, .@VIT;
			setunitdata .@gid, UMOB_INT, .@INT;
			setunitdata .@gid, UMOB_DEX, .@DEX;
			setunitdata .@gid, UMOB_LUK, .@LUK;
			
			setinstancevar("BOSS_ATK", .@MOB_ATK);
			setinstancevar("BOSS_ATK2", .@MOB_ATK2);
			setinstancevar("BOSS_MATK", .@MOB_MATK);
			setinstancevar("BOSS_MATK2", .@MOB_MATK2);
			setinstancevar("BOSS_STR", .@STR);
			setinstancevar("BOSS_AGI", .@AGI);
			setinstancevar("BOSS_INT", .@INT);
			setinstancevar("BOSS_VIT", .@VIT);
			setinstancevar("BOSS_DEX", .@DEX);
			setinstancevar("BOSS_LUK", .@LUK);
		}
		if( .@RES ) {
			.@RES = .@data[UMOB_RES] + (300 * .@RES);
			.@MRES = .@data[UMOB_MRES] + (300 * .@MRES);
			setunitdata .@gid,UMOB_RES,.@RES;
			setunitdata .@gid,UMOB_MRES,.@MRES;
			setinstancevar("BOSS_RES",.@RES);
			setinstancevar("BOSS_MRES",.@MRES);
		}
	}
	return;
}

function	script	ctower_revert_stats	{
	.@gid = getarg(0);
	if( unitexists(.@gid) ) {
		setunitdata .@gid, UMOB_ATKMIN, getinstancevar("BOSS_ATK");
		setunitdata .@gid, UMOB_ATKMAX, getinstancevar("BOSS_ATK2");
		setunitdata .@gid, UMOB_STR, getinstancevar("BOSS_STR");
		setunitdata .@gid, UMOB_AGI, getinstancevar("BOSS_AGI");
		setunitdata .@gid, UMOB_VIT, getinstancevar("BOSS_VIT");
		setunitdata .@gid, UMOB_INT, getinstancevar("BOSS_INT");
		setunitdata .@gid, UMOB_DEX, getinstancevar("BOSS_DEX");
		setunitdata .@gid, UMOB_LUK, getinstancevar("BOSS_LUK");
	}
	return;
}

function	script	ctower_recalc_stats	{
    .@gid = getarg(0);
    .@value = getarg(1);
    ctower_revert_stats(.@gid);
    if(unitexists(.@gid)){
        .@MOB_ATK = getinstancevar("BOSS_ATK") + (5000 * .@value);
        .@MOB_ATK2 = getinstancevar("BOSS_ATK2") + (5000 * .@value);
		.@MOB_MATK = getinstancevar("BOSS_MATK") + (5000 * .@value);
		.@MOB_MATK2 = getinstancevar("BOSS_MATK2") + (5000 * .@value);

        .@STR = getinstancevar("BOSS_STR") + (60 * .@value);
        .@AGI = getinstancevar("BOSS_AGI") + (60 * .@value);
        .@VIT = getinstancevar("BOSS_VIT") + (60 * .@value);
        .@INT = getinstancevar("BOSS_INT") + (60 * .@value);
        .@DEX = getinstancevar("BOSS_DEX") + (60 * .@value);
        .@LUK = getinstancevar("BOSS_LUK") + (60 * .@value);

		setunitdata .@gid, UMOB_ATKMIN, .@MOB_ATK;
		setunitdata .@gid, UMOB_ATKMAX, .@MOB_ATK2;
		setunitdata .@gid, UMOB_MATKMIN, .@MOB_MATK;
		setunitdata .@gid, UMOB_MATKMAX, .@MOB_MATK2;
		setunitdata .@gid, UMOB_STR, .@STR;
		setunitdata .@gid, UMOB_AGI, .@AGI;
		setunitdata .@gid, UMOB_VIT, .@VIT;
		setunitdata .@gid, UMOB_INT, .@INT;
		setunitdata .@gid, UMOB_DEX, .@DEX;
		setunitdata .@gid, UMOB_LUK, .@LUK;
    }
    return;
}
