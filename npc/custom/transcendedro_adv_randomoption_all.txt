// ============================================================
//  PHYSICAL NPC - สายกายภาพ
// ============================================================
grademk,102,55,6	script	Janna Physical#grademk	862,{
	disable_items;
	function rnd;
	mes "[Janna]";
	mes "Have you meet Brownie yet? She is my sister. She is so busy with some NEWBIES business, but you not newbies right?";
	next;
	mes "[Janna]";
	mes "Let me add some options to your high value items. I only charge you for the first time is ^00AA001,000,000 Zeny^000000. But if you want to reset it, surely, you have to pay another ^FF00005,000,000 Zeny^000000 for your next random options. ^^";
	next;
	switch(select("OK, let's do it.:Information.")) {
		case 1:
			mes "[Janna]";
			mes "";
			mes "You must equip the item, show me now!";
			for (.@i = 0; .@i < EQI_MAX; .@i++) {
				if (getequipisequiped(.@i)) {
					// แสดงทุกอุปกรณ์ที่สวมใส่เป็นสีเขียวหมด (รองรับทุกไอเทม)
					.@m1$ += "^11AD09" + F_getpositionname(.@i) + "-[" + getequipname(.@i) + "]^000000";
					.@exist = true;
				}
				.@m1$ += ":";
			}
			if (!.@exist) {
				mes "[Janna]";
				mes "";
				mes "You must equip the item, show me now!";
				end;
			}
			.@eq = select(.@m1$) - 1;
			clear;
			.@id = getequipid(.@eq);
			// ใช้ Group 1 สำหรับอุปกรณ์ทุกชนิด
			.@gid = 1;

			if (!getequiprandomoption(.@eq, 0, ROA_ID)) {
				.@zeny = .nz[.@gid];
				.@cashpoint = .nk[.@gid];
				.@costumpoint = .nc[.@gid];
				copyarray(.@item, getd(".ni_" + .@gid), getarraysize(getd(".ni_" + .@gid)));
				copyarray(.@amount, getd(".na_" + .@gid), getarraysize(getd(".na_" + .@gid)));
			} else {
				if (!.g_reroll[.@gid]) {
					mes "[Janna]";
					mes "";
					mes "This item cannot be used for further options!!";
					end;
				}
				.@zeny = .rz[.@gid];
				.@cashpoint = .rk[.@gid];
				.@costumpoint = .rc[.@gid];
				copyarray(.@item, getd(".ri_" + .@gid), getarraysize(getd(".ri_" + .@gid)));
				copyarray(.@amount, getd(".ra_" + .@gid), getarraysize(getd(".ra_" + .@gid)));
			}

			mes "Selected Item ^FF0000[" + getequipname(.@eq) + "]^000000";
			mes "==== Cost ====";
			if (.@zeny)
				mes "^FF0000Zeny: " + callfunc("F_InsertComma", .@zeny) + "^000000";
			if (.@cashpoint)
				mes "CashPoint: " + .@cashpoint;
			if (.@costumpoint)
				mes .CustomPointsName$ + ": " + .@costumpoint;
			for (.@i = 0; .@i < getarraysize(.@item); .@i++) {
				mes "(" + .@amount[.@i] + ") " + getitemname(.@item[.@i]);
			}
			if (!.KeepItemData) {
				mes "^E01E005If you persist, the upgrade fee and the cards you have installed may be lost forever.^000000";
			}
			mes "^00AA00Are you sure???^000000";
			if (select("S U R E !:Nope.") == 2) {
				clear;
				mes "Seeya!";
				end;
			}
			clear;

			if (Zeny < .@zeny) {
				mes "You don't have " + .@zeny + " Zeny";
				.@close = true;
			}
			if (#CASHPOINTS < .@cashpoint) {
				mes "You don't have " + .@cashpoint + " CashPoints";
				.@close = true;
			}
			if (getd(.CustomPointsVariable$) < .@costumpoint) {
				mes "You don't have " + .@costumpoint + " " + .CustomPointsName$;
				.@close = true;
			}
			for (.@i = 0; .@i < getarraysize(.@item); .@i++) {
				if (countitem(.@item[.@i]) < .@amount[.@i]) {
					mes "You don't have (" + .@amount[.@i] + ") " + getitemname(.@item[.@i]);
					.@close = true;
				}
			}
			if (.@close)
				end;

			.@r = getequiprefinerycnt(.@eq);
			.@grade = getenchantgrade(.@eq);
			for (.@i = 0; .@i < 4; .@i++) {
				if (getequipcardid(.@eq, .@i)) {
					setd ".@c" + .@i, getequipcardid(.@eq, .@i);
					.@re[getarraysize(.@re)] = true;
				}
			}

			copyarray(.@opt_id, getd(".ro_id_" + .@gid + "_0"), getarraysize(getd(".ro_id_" + .@gid + "_0")));
			copyarray(.@opt_mn, getd(".ro_mn_" + .@gid + "_0"), getarraysize(getd(".ro_mn_" + .@gid + "_0")));
			copyarray(.@opt_mx, getd(".ro_mx_" + .@gid + "_0"), getarraysize(getd(".ro_mx_" + .@gid + "_0")));
			for (.@i = 1; .@i < 6; .@i++) {
				copyarray(getd(".@opt_id" + .@i), getd(".ro_id_" + .@gid + "_" + .@i), getarraysize(getd(".ro_id_" + .@gid + "_" + .@i)));
				copyarray(getd(".@opt_mn" + .@i), getd(".ro_mn_" + .@gid + "_" + .@i), getarraysize(getd(".ro_mn_" + .@gid + "_" + .@i)));
				copyarray(getd(".@opt_mx" + .@i), getd(".ro_mx_" + .@gid + "_" + .@i), getarraysize(getd(".ro_mx_" + .@gid + "_" + .@i)));
			}

			for (.@i = 1; .@i < 6; .@i++) {
				if (rand(100) > getd(".s_" + .@i + "[" + .@gid + "]"))
					break;
				if (getd(".ro_id_" + .@gid + "_" + .@i)) {
					if (!getarraysize(getd(".@opt_id" + .@i)))
						break;
					.@ndx = rand(getarraysize(getd(".@opt_id" + .@i)));
					.@result_id = getd(".@opt_id" + .@i + "[" + .@ndx + "]");
					.@result_vl = rnd(getd(".@opt_mn" + .@i + "[" + .@ndx + "]"), getd(".@opt_mx" + .@i + "[" + .@ndx + "]"));
					deletearray(getd(".@opt_id" + .@i + "[" + .@ndx + "]"), 1);
					deletearray(getd(".@opt_mn" + .@i + "[" + .@ndx + "]"), 1);
					deletearray(getd(".@opt_mx" + .@i + "[" + .@ndx + "]"), 1);
				} else {
					if (!.@opt_id)
						break;
					.@ndx = rand(getarraysize(.@opt_id));
					.@result_id = .@opt_id[.@ndx];
					.@result_vl = rnd(.@opt_mn[.@ndx], .@opt_mx[.@ndx]);
					if (!.g_reuse[.@gid]) {
						deletearray(.@opt_id[.@ndx], 1);
						deletearray(.@opt_mn[.@ndx], 1);
						deletearray(.@opt_mx[.@ndx], 1);
					}
				}

				if (.ex[.@result_id]) {
					for (.@i2 = 0; .@i2 < getarraysize(.@r_id); .@i2++) {
						if (.ex[.@result_id] == .ex[.@r_id[.@i2]]) {
							.@reject = true;
						}
					}
				}

				if (!.@reject) {
					.@s1 = getarraysize(.@r_id);
					.@r_id[.@s1] = .@result_id;
					.@r_v[.@s1] = .@result_vl;
				} else {
					.@i--;
					.@failsafe++;
					if (.@failsafe > 50) {
						mes "Something went wrong , The server admin need to make sure that the config is done correctly!";
						end;
					}
				}
			}

			if (Zeny < .@zeny)
				end;
			if (#CASHPOINTS < .@cashpoint)
				end;
			if (getd(.CustomPointsVariable$) < .@costumpoint)
				end;
			for (.@i = 0; .@i < getarraysize(.@item); .@i++) {
				if (countitem(.@item[.@i]) < .@amount[.@i]) {
					end;
				}
			}

			// anti-hack
			if (callfunc("F_IsEquipIDHack", .@eq, .@id) || callfunc("F_IsEquipCardHack", .@eq, .@c0, .@c1, .@c2, .@c3)) {
				emotion ET_FRET;
				mes "Item Switch detected!";
				end;
			}

			Zeny -= .@zeny;
			#CASHPOINTS -= .@cashpoint;
			setd(.CustomPointsVariable$, getd(.CustomPointsVariable$) - .@costumpoint);
			for (.@i = 0; .@i < getarraysize(.@item); .@i++) {
				delitem(.@item[.@i], .@amount[.@i]);
			}

			if (!.@r_id) {
				specialeffect2 EF_REFINEOK;
				mes "Look Like you didn't get any random option, your luck is really bad!";
				end;
			}

			delequip(.@eq);
			specialeffect2 EF_REFINEOK;
			if (.KeepItemData)
				getitem4(.@id, 1, 1, .@r, 0, .@c0, .@c1, .@c2, .@c3, .@grade, .@r_id, .@r_v, .@null);
			else
				getitem4(.@id, 1, 1, 0, 0, 0, 0, 0, 0, 0, .@r_id, .@r_v, .@null);
			equip(.@id);
			specialeffect2 154;
			mes "Done!";
			end;

		case 2:
			mes "[Janna]";
			mes "I can add ^FF0000Physical Random Options^000000 to ^00AA00ALL equipment^000000 you are wearing!";
			mes "";
			mes "Just equip the item you want and talk to me.";
			mes "No need to register items anymore!";
			end;
	}
	end;

	function	rnd	{
		if (.HardBalance) {
			return rand(getarg(0), rand(getarg(0), getarg(1)));
		}
		return rand(getarg(0), getarg(1));
	}

	function	AddGroupOption	{
		.@id = getarg(0);
		.g_reroll[.@id] = getarg(1);
		.g_reuse[.@id] = getarg(2);
		return;
	}

	function	AddGroupCost	{
		.@id = getarg(0);
		.nz[.@id] = getarg(1);
		.nk[.@id] = getarg(2);
		.nc[.@id] = getarg(3);
		for (.@i = 4; .@i < getargcount(); .@i += 2) {
			.@s = getarraysize(getd(".ni_" + .@id));
			setd(".ni_" + .@id + "[" + .@s + "]", getarg(.@i));
			setd(".na_" + .@id + "[" + .@s + "]", getarg(.@i + 1));
		}
		return;
	}

	function	AddGroupReRollCost	{
		.@id = getarg(0);
		.rz[.@id] = getarg(1);
		.rk[.@id] = getarg(2);
		.rc[.@id] = getarg(3);
		for (.@i = 4; .@i < getargcount(); .@i += 2) {
			.@s = getarraysize(getd(".ri_" + .@id));
			setd(".ri_" + .@id + "[" + .@s + "]", getarg(.@i));
			setd(".ra_" + .@id + "[" + .@s + "]", getarg(.@i + 1));
		}
		return;
	}

	function	AddGroupChance	{
		.@id = getarg(0);
		.s_1[.@id] = getarg(1);
		.s_2[.@id] = getarg(2);
		.s_3[.@id] = getarg(3);
		.s_4[.@id] = getarg(4);
		.s_5[.@id] = getarg(5);
		return;
	}

	function	AddOpt	{
		.@id = getarg(0);
		.@loc = getarg(1);
		.@s = getarraysize(getd(".ro_id_" + .@id + "_" + .@loc));
		setd(".ro_id_" + .@id + "_" + .@loc + "[" + .@s + "]", getarg(2));
		if (getargcount() >= 5) {
			setd(".ro_mn_" + .@id + "_" + .@loc + "[" + .@s + "]", getarg(3));
			setd(".ro_mx_" + .@id + "_" + .@loc + "[" + .@s + "]", getarg(4));
		} else {
			setd(".ro_mn_" + .@id + "_" + .@loc + "[" + .@s + "]", 1);
			setd(".ro_mx_" + .@id + "_" + .@loc + "[" + .@s + "]", 1);
		}
		return;
	}

	function	AddExclusiveOpt	{
		.@s1 = getarraysize(.exv);
		.exv[.@s1] = true;
		for (.@i = 0; .@i < getargcount(); .@i++) {
			.ex[getarg(.@i)] = .@s1;
		}
		return;
	}

	OnInit:
		initnpctimer;
		.exv = true;
		.KeepItemData = true;
		.CustomPointsVariable$ = "#COSTUMPOINTS";
		.CustomPointsName$ = "Server RO Points";
		.HardBalance = false;

		// ===== CONFIGURATION FOR GROUP 1 (All Equipment) =====
		AddGroupOption(1, true, false);
		AddGroupCost(1, 1000000, 0, 0, 0, 0, 0, 0);
		AddGroupReRollCost(1, 1000000, 100, 0, 0, 0, 0, 0);
		AddGroupChance(1, 100, 85, 70, 50, 30);

		// SLOT #1 - Physical Stats
		AddOpt(1, 1, RDMOPT_VAR_STRAMOUNT, 5, 15);
		AddOpt(1, 1, RDMOPT_VAR_AGIAMOUNT, 5, 15);
		AddOpt(1, 1, RDMOPT_VAR_VITAMOUNT, 5, 15);
		AddOpt(1, 1, RDMOPT_VAR_DEXAMOUNT, 5, 15);
		AddOpt(1, 1, RDMOPT_VAR_LUKAMOUNT, 5, 15);
		AddOpt(1, 1, RDMOPT_VAR_ATTPOWER, 10, 50);
		AddOpt(1, 1, RDMOPT_VAR_ATKPERCENT, 5, 25);
		AddOpt(1, 1, RDMOPT_VAR_PLUSASPDPERCENT, 5, 15);
		AddOpt(1, 1, RDMOPT_VAR_HITSUCCESSVALUE, 5, 15);
		AddOpt(1, 1, RDMOPT_VAR_AVOIDSUCCESSVALUE, 5, 15);
		AddOpt(1, 1, RDMOPT_VAR_PLUSAVOIDSUCCESSVALUE, 5, 15);
		AddOpt(1, 1, RDMOPT_VAR_CRITICAL_RATE, 5, 15);
		AddOpt(1, 1, RDMOPT_DEC_SPELL_DELAY_TIME, 5, 15);

		// SLOT #2 - Physical Damage
		AddOpt(1, 2, RDMOPT_VAR_POWAMOUNT, 3, 10);
		AddOpt(1, 2, RDMOPT_VAR_STAAMOUNT, 3, 10);
		AddOpt(1, 2, RDMOPT_VAR_CONAMOUNT, 3, 10);
		AddOpt(1, 2, RDMOPT_VAR_CRTAMOUNT, 3, 10);
		AddOpt(1, 2, RDMOPT_VAR_PATKAMOUNT, 5, 15);
		AddOpt(1, 2, RDMOPT_DAMAGE_CRI_TARGET, 5, 25);
		AddOpt(1, 2, RDMOPT_MELEE_ATTACK_DAMAGE_TARGET, 5, 25);
		AddOpt(1, 2, RDMOPT_RANGE_ATTACK_DAMAGE_TARGET, 5, 25);
		AddOpt(1, 2, RDMOPT_NON_CRIT_ATK_RATE, 5, 25);
		AddOpt(1, 2, RDMOPT_VAR_CRITICALSUCCESSVALUE, 5, 15);

		// SLOT #3 - Element & Race Damage
		AddOpt(1, 3, RDMOPT_DAMAGE_PROPERTY_NOTHING_TARGET, 5, 25);
		AddOpt(1, 3, RDMOPT_DAMAGE_PROPERTY_WATER_TARGET, 5, 25);
		AddOpt(1, 3, RDMOPT_DAMAGE_PROPERTY_GROUND_TARGET, 5, 25);
		AddOpt(1, 3, RDMOPT_DAMAGE_PROPERTY_FIRE_TARGET, 5, 25);
		AddOpt(1, 3, RDMOPT_DAMAGE_PROPERTY_WIND_TARGET, 5, 25);
		AddOpt(1, 3, RDMOPT_DAMAGE_PROPERTY_POISON_TARGET, 5, 25);
		AddOpt(1, 3, RDMOPT_DAMAGE_PROPERTY_SAINT_TARGET, 5, 25);
		AddOpt(1, 3, RDMOPT_DAMAGE_PROPERTY_DARKNESS_TARGET, 5, 25);
		AddOpt(1, 3, RDMOPT_DAMAGE_PROPERTY_TELEKINESIS_TARGET, 5, 25);
		AddOpt(1, 3, RDMOPT_DAMAGE_PROPERTY_UNDEAD_TARGET, 5, 25);
		AddOpt(1, 3, RDMOPT_RACE_DAMAGE_NOTHING, 5, 25);
		AddOpt(1, 3, RDMOPT_RACE_DAMAGE_UNDEAD, 5, 25);
		AddOpt(1, 3, RDMOPT_RACE_DAMAGE_ANIMAL, 5, 25);
		AddOpt(1, 3, RDMOPT_RACE_DAMAGE_PLANT, 5, 25);
		AddOpt(1, 3, RDMOPT_RACE_DAMAGE_INSECT, 5, 25);
		AddOpt(1, 3, RDMOPT_RACE_DAMAGE_FISHS, 5, 25);
		AddOpt(1, 3, RDMOPT_RACE_DAMAGE_DEVIL, 5, 25);
		AddOpt(1, 3, RDMOPT_RACE_DAMAGE_HUMAN, 5, 25);
		AddOpt(1, 3, RDMOPT_RACE_DAMAGE_ANGEL, 5, 25);
		AddOpt(1, 3, RDMOPT_RACE_DAMAGE_DRAGON, 5, 25);
		AddOpt(1, 3, RDMOPT_DAMAGE_SIZE_PERFECT, 0, 1);
		AddOpt(1, 3, RDMOPT_BODY_INDESTRUCTIBLE, 0, 1);

		// SLOT #4 - Size & Defense
		AddOpt(1, 4, RDMOPT_DAMAGE_SIZE_SMALL_TARGET, 5, 25);
		AddOpt(1, 4, RDMOPT_DAMAGE_SIZE_MIDIUM_TARGET, 5, 25);
		AddOpt(1, 4, RDMOPT_DAMAGE_SIZE_LARGE_TARGET, 5, 25);
		AddOpt(1, 4, RDMOPT_VAR_MAXHPPERCENT, 5, 20);
		AddOpt(1, 4, RDMOPT_VAR_MAXSPPERCENT, 5, 20);
		AddOpt(1, 4, RDMOPT_VAR_RESAMOUNT, 5, 15);
		AddOpt(1, 4, RDMOPT_VAR_MRESAMOUNT, 5, 15);
		AddOpt(1, 4, RDMOPT_REFLECT_DAMAGE_PERCENT, 5, 15);

		// SLOT #5 - EXP & Class Damage
		AddOpt(1, 5, RDMOPT_ADDEXPPERCENT_KILLRACE_ALL, 5, 15);
		AddOpt(1, 5, RDMOPT_CLASS_DAMAGE_NORMAL_TARGET, 5, 25);
		AddOpt(1, 5, RDMOPT_CLASS_DAMAGE_BOSS_TARGET, 5, 25);
	end;

	OnTimer1000:
		showscript(" ! Random Opt Physical ! ");
	end;
	OnTimer4000:
		showscript(" ! Random Opt Physical ! ");
	end;
	OnTimer7000:
		showscript(" ! Random Opt Physical ! ");
	end;
	OnTimer10000:
		specialeffect EF_TETRACASTING;
		setnpctimer 0;
	end;
}

			copyarray(.@opt_id, getd(".ro_id_" + .@gid + "_0"), getarraysize(getd(".ro_id_" + .@gid + "_0")));
			copyarray(.@opt_mn, getd(".ro_mn_" + .@gid + "_0"), getarraysize(getd(".ro_mn_" + .@gid + "_0")));
			copyarray(.@opt_mx, getd(".ro_mx_" + .@gid + "_0"), getarraysize(getd(".ro_mx_" + .@gid + "_0")));
			for (.@i = 1; .@i < 6; .@i++) {
				copyarray(getd(".@opt_id" + .@i), getd(".ro_id_" + .@gid + "_" + .@i), getarraysize(getd(".ro_id_" + .@gid + "_" + .@i)));
				copyarray(getd(".@opt_mn" + .@i), getd(".ro_mn_" + .@gid + "_" + .@i), getarraysize(getd(".ro_mn_" + .@gid + "_" + .@i)));
				copyarray(getd(".@opt_mx" + .@i), getd(".ro_mx_" + .@gid + "_" + .@i), getarraysize(getd(".ro_mx_" + .@gid + "_" + .@i)));
			}

			for (.@i = 1; .@i < 6; .@i++) {
				if (rand(100) > getd(".s_" + .@i + "[" + .@gid + "]"))
					break;
				if (getd(".ro_id_" + .@gid + "_" + .@i)) {
					if (!getarraysize(getd(".@opt_id" + .@i)))
						break;
					.@ndx = rand(getarraysize(getd(".@opt_id" + .@i)));
					.@result_id = getd(".@opt_id" + .@i + "[" + .@ndx + "]");
					.@result_vl = rnd(getd(".@opt_mn" + .@i + "[" + .@ndx + "]"), getd(".@opt_mx" + .@i + "[" + .@ndx + "]"));
					deletearray(getd(".@opt_id" + .@i + "[" + .@ndx + "]"), 1);
					deletearray(getd(".@opt_mn" + .@i + "[" + .@ndx + "]"), 1);
					deletearray(getd(".@opt_mx" + .@i + "[" + .@ndx + "]"), 1);
				} else {
					if (!.@opt_id)
						break;
					.@ndx = rand(getarraysize(.@opt_id));
					.@result_id = .@opt_id[.@ndx];
					.@result_vl = rnd(.@opt_mn[.@ndx], .@opt_mx[.@ndx]);
					if (!.g_reuse[.@gid]) {
						deletearray(.@opt_id[.@ndx], 1);
						deletearray(.@opt_mn[.@ndx], 1);
						deletearray(.@opt_mx[.@ndx], 1);
					}
				}

				if (.ex[.@result_id]) {
					for (.@i2 = 0; .@i2 < getarraysize(.@r_id); .@i2++) {
						if (.ex[.@result_id] == .ex[.@r_id[.@i2]]) {
							.@reject = true;
						}
					}
				}

				if (!.@reject) {
					.@s1 = getarraysize(.@r_id);
					.@r_id[.@s1] = .@result_id;
					.@r_v[.@s1] = .@result_vl;
				} else {
					.@i--;
					.@failsafe++;
					if (.@failsafe > 50) {
						mes "Something went wrong , The server admin need to make sure that the config is done correctly!";
						end;
					}
				}
			}

			if (Zeny < .@zeny)
				end;
			if (#CASHPOINTS < .@cashpoint)
				end;
			if (getd(.CustomPointsVariable$) < .@costumpoint)
				end;
			for (.@i = 0; .@i < getarraysize(.@item); .@i++) {
				if (countitem(.@item[.@i]) < .@amount[.@i]) {
					end;
				}
			}

			// anti-hack
			if (callfunc("F_IsEquipIDHack", .@eq, .@id) || callfunc("F_IsEquipCardHack", .@eq, .@c0, .@c1, .@c2, .@c3)) {
				emotion ET_FRET;
				mes "Item Switch detected!";
				end;
			}

			Zeny -= .@zeny;
			#CASHPOINTS -= .@cashpoint;
			setd(.CustomPointsVariable$, getd(.CustomPointsVariable$) - .@costumpoint);
			for (.@i = 0; .@i < getarraysize(.@item); .@i++) {
				delitem(.@item[.@i], .@amount[.@i]);
			}

			if (!.@r_id) {
				specialeffect2 EF_REFINEOK;
				mes "Look Like you didn't get any random option, your luck is really bad!";
				end;
			}

			delequip(.@eq);
			specialeffect2 EF_REFINEOK;
			if (.KeepItemData)
				getitem4(.@id, 1, 1, .@r, 0, .@c0, .@c1, .@c2, .@c3, .@grade, .@r_id, .@r_v, .@null);
			else
				getitem4(.@id, 1, 1, 0, 0, 0, 0, 0, 0, 0, .@r_id, .@r_v, .@null);
			equip(.@id);
			specialeffect2 154;
			mes "Done!";
			end;

		case 2:
			mes "[Janna]";
			mes "I can add ^FF0000Magical Random Options^000000 to ^00AA00ALL equipment^000000 you are wearing!";
			mes "";
			mes "Just equip the item you want and talk to me.";
			mes "No need to register items anymore!";
			end;
	}
	end;

	function	rnd	{
		if (.HardBalance) {
			return rand(getarg(0), rand(getarg(0), getarg(1)));
		}
		return rand(getarg(0), getarg(1));
	}

		.@id = getarg(0);

		.@id = getarg(0);
		.@loc = getarg(1);

			setd(".ro_mx_" + .@id + "_" + .@loc + "[" + .@s + "]", getarg(4));

			setd(".ro_mx_" + .@id + "_" + .@loc + "[" + .@s + "]", 1);
		}
		return;
	}

		}
		return;
	}
