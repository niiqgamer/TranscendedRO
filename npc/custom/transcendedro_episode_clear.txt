function	script	F_EpDesc	{
	.@ep = getarg(0);
	mes "[Balheele]";
	mes "^4d4dffEpisode " + .@ep + "^000000?";
	switch (.@ep) {
		case 16: mes "It's a story surrounding Terra Gloria."; break;
		case 17: mes "It's a story about Illusion and Varmunt."; break;
		case 18: mes "It's a story related to the Gray Wolf Village."; break;
		case 19: mes "It's a story related to Isgard's Ice Castle."; break;
	}
	mes "Let me take a look.";
	next;
	return;
}

function	script	F_CheckQuestState	{
	.@id = getarg(0);
	.@state = checkquest(.@id, PLAYTIME);
	if (.@state == 2) return "^4d4dffCompleted^000000";
	return "Incomplete";
}

// Returns true when the player finished this episode through official story scripts
// (variables / story quest IDs), without Balheele chapter marker quests.
function	script	F_EpOrganicComplete	{
	.@ep = getarg(0);
	.@part = getarg(1, 0);

	switch (.@ep) {
	case 16:
		return (terra_gloria_main >= 25);
	case 17:
		if (.@part == 2)
			return (isbegin_quest(8681) == 2 || ep17_2_main >= 36);
		return (isbegin_quest(16360) == 2 || ep17_2_main >= 1);
	case 18:
		return (ep18_main >= 57);
	case 19:
		return (ep19_main >= 100);
	}
	return 0;
}

// Prior-episode prerequisite: chapter marker quest OR organic story completion.
function	script	F_EpPrereqMet	{
	.@marker_id = getarg(0);

	if (!.@marker_id)
		return 1;
	if (checkquest(.@marker_id, PLAYTIME) == 2)
		return 1;

	switch (.@marker_id) {
	case 7741: return callfunc("F_EpOrganicComplete", 16, 0);
	case 16360: return callfunc("F_EpOrganicComplete", 17, 1);
	case 8681: return callfunc("F_EpOrganicComplete", 17, 2);
	case 18085: return callfunc("F_EpOrganicComplete", 18, 0);
	}
	return 0;
}

function	script	F_EpEpisodeComplete	{
	.@ep = getarg(0);
	.@part = getarg(1, 0);
	.@marker_id = getarg(2, 0);

	if (.@marker_id && checkquest(.@marker_id, PLAYTIME) == 2)
		return 1;
	return callfunc("F_EpOrganicComplete", .@ep, .@part);
}

function	script	F_EpForceComplete	{
	.@id = getarg(0);

	if (checkquest(.@id) != -1)
		erasequest .@id;

	setquest .@id;

	.@state = checkquest(.@id);
	if (.@state >= 0 && .@state < 2)
		completequest .@id;
	return;
}

function	script	F_EpPrereqFail	{
	.@ep = getarg(0);
	.@part = getarg(1, 0);
	.@require_id = getarg(2, 0);
	.@area$ = getarg(3, "");

	if (callfunc("F_EpPrereqMet", .@require_id))
		return 0;

	mes "[Balheele]";
	mes "There's an order to organizing every story.";
	if (.@area$ != "")
		mes "You haven't collected the story related to ^4d4dff" + .@area$ + "^000000 yet.";
	else
		mes "You haven't completed the required earlier episode yet.";
	next;
	mes "[Balheele]";
	if (.@ep == 17 && .@part == 2)
		mes "Could you come back after ^4d4dffcompleting Episode 17.1^000000?";
	else if (.@ep > 13)
		mes "Could you come back after ^4d4dffcompleting Episode " + (.@ep - 1) + "^000000?";
	else
		mes "Could you come back after ^4d4dffcompleting the required earlier episode^000000?";
	close;
	return 1;
}

	switch (.@ep) {
	case 16:
		if (terra_gloria_main < 25)
			set terra_gloria_main, 25;
		if (ep16_royal < 24)
			set ep16_royal, 24;
		break;

	case 17:
		if (.@part == 2) {
			if (ep17_2_main < 36)
				set ep17_2_main, 36;
			if (ep172_watergarden < 3)
				set ep172_watergarden, 3;
		} else {
			if (ep17_2_main < 1)
				set ep17_2_main, 1;
		}
		break;

	case 18:
		if (ep18_main < 57)
			set ep18_main, 57;
		break;

	case 19:
		if (ep19_main < 100)
			set ep19_main, 100;
		break;
	}

	// If hidden NPC repair is implemented later, enable this line.
	// callfunc("F_EpRepairHiddenNpcs");

	return;
