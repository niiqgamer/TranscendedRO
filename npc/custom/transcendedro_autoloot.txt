// ==========================================================
// Autoloot Configuration Menu
// ==========================================================
function	script	F_AutolootMenu	{
L_MainMenu:
	// ใช้ตัวแปรคาแรคเตอร์แบบถาวร (ไม่มี @ นำหน้า) เพื่อจำสถานะเปิด/ปิด
	if (Autoloot_ON != 1) {
		Autoloot_ON = 0;
	}

	// แสดงผลสถานะ
	if (Autoloot_ON == 1) {
		.@autoloot_status$ = "^00CC00[ON]^000000";
	} else {
		.@autoloot_status$ = "^FF0000[OFF]^000000";
	}

	mes "[ Autoloot Manager ]";
	mes "Autoloot Configuration System";
	mes "You can set up to 10 Types and 10 IDs";
	mes "-----------------------------------";
	mes "Autoloot Status: " + .@autoloot_status$;
	mes "Types Set: " + @aloot_type_count + " / 10";
	mes "IDs Set: " + @aloot_id_count + " / 10";
	mes "-----------------------------------";
	next;

	switch(select(
		"Toggle Autoloot (" + .@autoloot_status$ + ")", 
		"Manage Autoloot Type (@aloottype)", 
		"Manage Autoloot ID (@alootid)", 
		"Clear All Settings", 
		"Close Menu"
	)) {
		case 1:
			if (Autoloot_ON == 1) {
				atcommand "@autoloot 0";
				Autoloot_ON = 0;
				dispbottom "Autoloot has been disabled.";
			} else {
				atcommand "@autoloot 100"; // ตั้งค่าเริ่มต้นเก็บ 100% ตอนเปิด
				Autoloot_ON = 1;
				dispbottom "Autoloot has been enabled (100%).";
			}
			next;
			goto L_MainMenu;
			
		case 2:
			callsub L_ManageTypes;
			goto L_MainMenu;
			
		case 3:
			callsub L_ManageIDs;
			goto L_MainMenu;
			
		case 4:
			// รีเซ็ตการตั้งค่าทั้งหมด
			atcommand "@aloottype reset";
			atcommand "@alootid reset";
			atcommand "@autoloot 0";
			
			// ล้างตัวแปรที่ใช้นับและเก็บข้อมูลทั้งหมด
			for (.@i = 0; .@i < 10; .@i++) {
				setd "@aloot_type_" + .@i, 0;
				setd "@aloot_id_" + .@i, 0;
			}
			@aloot_type_count = 0;
			@aloot_id_count = 0;
			Autoloot_ON = 0;
			
			mes "[ Autoloot Manager ]";
			mes "All Autoloot settings cleared successfully!";
			mes "-----------------------------------";
			mes "- Executed: @aloottype reset";
			mes "- Executed: @alootid reset";
			mes "- Autoloot Status set to: ^FF0000OFF^000000";
			next;
			goto L_MainMenu;
			
		case 5:
			close;
	}
	end;

// ----------------------------------------------------------
// Sub: Manage Autoloot Type
// ----------------------------------------------------------
L_ManageTypes:
	mes "[ Manage Autoloot Type ]";
	mes "Select item type to auto-loot";
	mes "(Selecting an active type will disable it)";
	mes "-----------------------------------";
	mes "Current Types Configured (" + @aloot_type_count + " / 10):";
	if (@aloot_type_count == 0) {
		mes " - None";
	} else {
		for (.@i = 0; .@i < @aloot_type_count; .@i++) {
			mes " -> Type: " + getd("@aloot_type_" + .@i);
		}
	}
	mes "-----------------------------------";
	next;
	
	switch(select(
		"Healing (0)", 
		"Usable (2)", 
		"Etc (3)", 
		"Weapon (4)", 
		"Armor (5)", 
		"Card (6)", 
		"Pet Egg (7)", 
		"Pet Armor (8)", 
		"Ammo (10)", 
		"Enter Custom Type", 
		"Go Back"
	)) {
		case 1:  .@type = 0; break;
		case 2:  .@type = 2; break;
		case 3:  .@type = 3; break;
		case 4:  .@type = 4; break;
		case 5:  .@type = 5; break;
		case 6:  .@type = 6; break;
		case 7:  .@type = 7; break;
		case 8:  .@type = 8; break;
		case 9:  .@type = 10; break;
		case 10:
			mes "Enter the Item Type Number:";
			input .@type;
			if (.@type < 0) { 
				mes "Invalid type number."; 
				next; 
				return; 
			}
			break;
		case 11: 
			return;
	}

	// ตรวจสอบว่าแอดประเภทนี้ไปแล้วหรือยัง
	.@found = 0;
	for (.@i = 0; .@i < @aloot_type_count; .@i++) {
		if (getd("@aloot_type_" + .@i) == .@type) {
			.@found = 1;
			.@idx = .@i;
			break;
		}
	}

	if (.@found) {
		// ถ้าเจอแล้ว ให้ทำการปิด (ลบออก)
		atcommand "@aloottype -" + .@type;
		for (.@i = .@idx; .@i < @aloot_type_count - 1; .@i++) {
			setd "@aloot_type_" + .@i, getd("@aloot_type_" + (.@i + 1));
		}
		setd "@aloot_type_" + (@aloot_type_count - 1), 0;
		@aloot_type_count--;
		dispbottom "Removed Item Type " + .@type + " from Autoloot.";
	} else {
		// ถ้ายังไม่เจอ ให้เช็คว่าเกินโควต้า 10 ชนิดไหม ถ้าไม่เกินก็แอดเข้าเพิ่ม
		if (@aloot_type_count >= 10) {
			mes "You cannot configure more than 10 Types!";
			next;
			return;
		}
		atcommand "@aloottype +" + .@type;
		setd "@aloot_type_" + @aloot_type_count, .@type;
		@aloot_type_count++;
		dispbottom "Added Item Type " + .@type + " to Autoloot.";
	}
	return;

// ----------------------------------------------------------
// Sub: Manage Autoloot ID
// ----------------------------------------------------------
L_ManageIDs:
	mes "[ Manage Autoloot ID ]";
	mes "Configure specific items to auto-loot";
	mes "(Selecting/Entering an active ID will disable it)";
	mes "-----------------------------------";
	mes "Current IDs Configured (" + @aloot_id_count + " / 10):";
	if (@aloot_id_count == 0) {
		mes " - None";
	} else {
		for (.@i = 0; .@i < @aloot_id_count; .@i++) {
			.@current_id = getd("@aloot_id_" + .@i);
			mes " -> " + .@current_id + " [" + getitemname(.@current_id) + "]";
		}
	}
	mes "-----------------------------------";
	next;
	
	switch(select("Add/Remove Item ID", "Go Back")) {
		case 1:
			mes "Enter Item ID:";
			input .@id;
			if (.@id <= 0 || getitemname(.@id) == "null" || getitemname(.@id) == "") {
				mes "Invalid Item ID or Item does not exist.";
				next;
				return;
			}
			break;
		case 2:
			return;
	}

	// ตรวจสอบว่าแอด ID นี้ไปแล้วหรือยัง
	.@found = 0;
	for (.@i = 0; .@i < @aloot_id_count; .@i++) {
		if (getd("@aloot_id_" + .@i) == .@id) {
			.@found = 1;
			.@idx = .@i;
			break;
		}
	}

	if (.@found) {
		// ถ้าเจอแล้วให้ ลบออก
		atcommand "@alootid -" + .@id;
		for (.@i = .@idx; .@i < @aloot_id_count - 1; .@i++) {
			setd "@aloot_id_" + .@i, getd("@aloot_id_" + (.@i + 1));
		}
		setd "@aloot_id_" + (@aloot_id_count - 1), 0;
		@aloot_id_count--;
		dispbottom "Removed item: " + getitemname(.@id) + " (" + .@id + ") from Autoloot.";
	} else {
		// เช็คโควต้า ID
		if (@aloot_id_count >= 10) {
			mes "You cannot configure more than 10 Item IDs!";
			next;
			return;
		}
		atcommand "@alootid +" + .@id;
		setd "@aloot_id_" + @aloot_id_count, .@id;
		@aloot_id_count++;
		dispbottom "Added item: " + getitemname(.@id) + " (" + .@id + ") to Autoloot.";
	}
	return;
}
