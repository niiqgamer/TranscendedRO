-	script	Dice_System	-1,{

OnInit:
	// คำสั่งที่ผู้เล่นใช้ได้ = @dice = วาร์ปเข้ามาห้องทอยเต๋า
	bindatcmd "dice", strnpcinfo(0)+"::OnToDiceEvent", 0;
	
	// @dicedooropen = เปิดประตูเลือกหน้าเต๋า *GM ใช้ได้เท่านั้น
	bindatcmd "dicedooropen", strnpcinfo(0)+"::OnFenceOff", 99;
	
	// @dicedoorclose = ปิดประตูเลือกหน้าเต๋า *GM ใช้ได้เท่านั้น
	bindatcmd "dicedoorclose", strnpcinfo(0)+"::OnFenceOn", 99;
	
	// ห้ามเดินทั้งแมพ
	setcell "quiz_02", 286, 269, 323, 232, cell_walkable, 0;
	
	// เดินได้แค่ทางเดินกลาง
	setcell "quiz_02", 302, 264, 305, 237, cell_walkable, 1;
	
	disablenpc "Dice_Event";
	hideonnpc "Reward for Dice";
	hideonnpc "Dice Manager";
	for (.@i = 1; .@i <= 6; .@i++) {
		enablenpc("WT#" + .@i);
		hideonnpc("WT#" + .@i);
	}
	for (.@i = 1; .@i <= 6; .@i++) {
		enablenpc("Effect_NPC#" + .@i);
		hideonnpc("Effect_NPC#" + .@i);
	}
	end;

OnToDiceEvent:
	if (.DiceEvent == 0) {
		mes "[Dice Event]";
		mes " ";
		mes "^AA0000Event has not yet begin,^000000";
		mes "^AA0000or event is over now, sorry.^000000";
		mes " ";
		mes "Follow up the event at Discord.";
		close;
		end;
	} else if (.DiceEvent == 2) {
		mes "[Dice Event]";
		mes " ";
		mes "^AA0000Sorry for that,^000000";
		mes "^AA0000Dice's entrance is closed.^000000";
		mes " ";
		mes "Follow up the event at Discord.";
		close;
		end;
	} else if (.DiceEvent == 1) {
		warp "quiz_02",303,251;
		end;
	}
	end;

OnFenceOff:
	setcell "quiz_02", 289, 263, 300, 256, cell_walkable, 1;						// พื้นที่ 1
	setcell "quiz_02", 301, 260, 301, 259, cell_walkable, 1;						// ประตู 1
	hideonnpc "Wall#018";																			// ประตู 1
	hideonnpc "Wall#019";																			// ประตู 1
	setcell "quiz_02", 289, 254, 300, 247, cell_walkable, 1;						// พื้นที่ 2
	setcell "quiz_02", 301, 251, 301, 250, cell_walkable, 1;						// ประตู 2
	hideonnpc "Wall#027";																			// ประตู 2
	hideonnpc "Wall#028";																			// ประตู 2
	setcell "quiz_02", 289, 245, 300, 236, cell_walkable, 1;						// พื้นที่ 3
	setcell "quiz_02", 301, 242, 301, 241, cell_walkable, 1;						// ประตู 3
	hideonnpc "Wall#036";																			// ประตู 3
	hideonnpc "Wall#037";																			// ประตู 3
	setcell "quiz_02", 307, 263, 318, 256, cell_walkable, 1;						// พื้นที่ 4
	setcell "quiz_02", 306, 260, 306, 259, cell_walkable, 1;						// ประตู 4
	hideonnpc "Wall#099";																			// ประตู 4
	hideonnpc "Wall#100";																			// ประตู 4
	setcell "quiz_02", 307, 254, 318, 247, cell_walkable, 1;						// พื้นที่ 5
	setcell "quiz_02", 306, 251, 306, 250, cell_walkable, 1;						// ประตู 5
	hideonnpc "Wall#108";																			// ประตู 5
	hideonnpc "Wall#109";																			// ประตู 5
	setcell "quiz_02", 307, 245, 318, 238, cell_walkable, 1;						// พื้นที่ 6
	setcell "quiz_02", 306, 242, 306, 241, cell_walkable, 1;						// ประตู 6
	hideonnpc "Wall#117";																			// ประตู 6
	hideonnpc "Wall#118";																			// ประตู 6
	end;

OnFenceOn:
	setcell "quiz_02", 289, 263, 300, 256, cell_walkable, 1;						// พื้นที่ 1
	setcell "quiz_02", 301, 260, 301, 259, cell_walkable, 0;						// ประตู 1
	hideoffnpc "Wall#018";																			// ประตู 1
	hideoffnpc "Wall#019";																			// ประตู 1
	setcell "quiz_02", 289, 254, 300, 247, cell_walkable, 1;						// พื้นที่ 2
	setcell "quiz_02", 301, 251, 301, 250, cell_walkable, 0;						// ประตู 2
	hideoffnpc "Wall#027";																			// ประตู 2
	hideoffnpc "Wall#028";																			// ประตู 2
	setcell "quiz_02", 289, 245, 300, 236, cell_walkable, 1;						// พื้นที่ 3
	setcell "quiz_02", 301, 242, 301, 241, cell_walkable, 0;						// ประตู 3
	hideoffnpc "Wall#036";																			// ประตู 3
	hideoffnpc "Wall#037";																			// ประตู 3
	setcell "quiz_02", 307, 263, 318, 256, cell_walkable, 1;						// พื้นที่ 4
	setcell "quiz_02", 306, 260, 306, 259, cell_walkable, 0;						// ประตู 4
	hideoffnpc "Wall#099";																			// ประตู 4
	hideoffnpc "Wall#100";																			// ประตู 4
	setcell "quiz_02", 307, 254, 318, 247, cell_walkable, 1;						// พื้นที่ 5
	setcell "quiz_02", 306, 251, 306, 250, cell_walkable, 0;						// ประตู 5
	hideoffnpc "Wall#108";																			// ประตู 5
	hideoffnpc "Wall#109";																			// ประตู 5
	setcell "quiz_02", 307, 245, 318, 238, cell_walkable, 1;						// พื้นที่ 6
	setcell "quiz_02", 306, 242, 306, 241, cell_walkable, 0;						// ประตู 6
	hideoffnpc "Wall#117";																			// ประตู 6
	hideoffnpc "Wall#118";																			// ประตู 6
	end;

OnStart:
	.DiceEvent = 1;
	enablenpc "Dice_Event";
	donpcevent "Dice_System::OnFenceOff";
	announce "@dice : DICE EVENT IS ON!!! Typing ' @dice ' to join the event!!!",bc_all,0xEA0000,FW_BOLD,20,0,0,0;
	sleep 10000;
	announce "@dice : You have only 2 minutes for joining us ... HURRY!",bc_all,0xEA0000,FW_BOLD,15,0,0,0;
	sleep 40000;
	announce "@dice : DICE EVENT IS ON!!! Typing ' @dice ' to join the event!!!",bc_all,0xEA0000,FW_BOLD,20,0,0,0;
	sleep 40000;
	announce "@dice : You have only 40 seconds for joining us ... HURRY!",bc_all,0xEA0000,FW_BOLD,15,0,0,0;
	sleep 35000;
	announce "@dice : You have ONLY 10 seconds!!!",bc_all,0xEA0000,FW_BOLD,15,0,0,0;
	sleep 5000;
	announce "@dice :  5...",bc_all,0xEA0000,FW_BOLD,15,0,0,0;
	sleep 1000;
	announce "@dice :  4...",bc_all,0xEA0000,FW_BOLD,15,0,0,0;
	sleep 1000;
	announce "@dice :  3...",bc_all,0xEA0000,FW_BOLD,15,0,0,0;
	sleep 1000;
	announce "@dice :  2...",bc_all,0xEA0000,FW_BOLD,15,0,0,0;
	sleep 1000;
	announce "@dice :  1...",bc_all,0xEA0000,FW_BOLD,15,0,0,0;
	sleep 1000;
	announce "Time is up!!! Dice's entrance is closed!",bc_all,0xEA0000,FW_BOLD,20,0,0,0;
	.DiceEvent = 2;
	disablenpc "Dice_Event";
	donpcevent "Dice_System::OnStartDice";
	end;
	
OnStartDice:
	hideoffnpc "Reward for Dice";

	// กำหนดค่าต่าง ๆ
	setarray .areas[0], 289,263, 300,256;
	setarray .areas[1], 289,254, 300,247;
	setarray .areas[2], 289,245, 300,236;
	setarray .areas[3], 307,263, 318,256;
	setarray .areas[4], 307,254, 318,247;
	setarray .areas[5], 307,245, 318,238;

	setarray .remaining_areas, 1, 2, 3, 4, 5, 6; // พื้นที่ที่เหลือทั้งหมด

	sleep 5000;
	mapannounce "quiz_02", "@dice : You have only 1 minute to choose your room. Let's go!", bc_map, 0xEA0000, FW_BOLD, 15, 0, 0;
	sleep 30000;
	mapannounce "quiz_02", "@dice : Every room will closed in 30 seconds!!!", bc_map, 0xEA0000, FW_BOLD, 15, 0, 0;
	sleep 30000;
	mapannounce "quiz_02", "@dice : Time is up!!!", bc_map, 0xEA0000, FW_BOLD, 20, 0, 0;
	donpcevent "Dice_System::OnFenceOn";

	if (getmapusers("quiz_02") <= 1) {
		mapannounce "quiz_02", "@dice : So sorry, event canceled due to insufficient number of players.", bc_map, 0xEA0000, FW_BOLD, 15, 0, 0;
		sleep 5000;
		mapwarp "quiz_02", "izlude", 128, 130;
		.DiceEvent = 0;
		end;
	}
	if (getmapusers("quiz_02") <= 3) {
		mapannounce "quiz_02", "@dice : Event canceled due to having fewer than 4 players. Please go claim your reward.", bc_map, 0xEA0000, FW_BOLD, 15, 0, 0;
		sleep 5000;
		mapwarp "quiz_02", "sec_in01", 168, 173;
		.DiceEvent = 0;
		end;
	}

	.@rounds = 0;
	while (.@rounds < 5) {
		sleep 10000;
		mapannounce "quiz_02", "DICE! Round # " + (.@rounds + 1) + "!!!", bc_map, 0xEA0000, FW_BOLD, 50, 0, 0;
		sleep 2000;
		
		// เริ่ม Effect
		.@effectSpeed = 1;
		.@loopCount = rand(110, 130);
		.@currentIndex = 0;

		while (.@loopCount > 0) {
			// เลือกพื้นที่ถัดไป (หมุนเวียนเฉพาะพื้นที่ที่เหลือ)
			.@currentIndex = (.@currentIndex + 1) % getarraysize(.remaining_areas);

			// เปิด Effect ที่พื้นที่ใหม่
			specialeffect 636, AREA, "Effect_NPC#" + .remaining_areas[.@currentIndex];

			sleep .@effectSpeed;
			.@effectSpeed += 4;
			.@loopCount--;
		}

		// สุ่มหาพื้นที่ที่จะถูกคัดออก
		.@idx = rand(0, getarraysize(.remaining_areas) - 1);
		.@area_idx = .remaining_areas[.@idx];

		// หมุน Effect ไปยังพื้นที่ที่ถูกคัดออก ??
		while (.remaining_areas[.@currentIndex] != .@area_idx) {
			// ขยับไปพื้นที่ถัดไป
			.@currentIndex = (.@currentIndex + 1) % getarraysize(.remaining_areas);

			// เปิด Effect ที่พื้นที่ใหม่
			specialeffect 636, AREA, "Effect_NPC#" + .remaining_areas[.@currentIndex];

			// เพิ่มความเร็วในการหมุนช้า ๆ ก่อนหยุด
			sleep 150;
		}

		// ลบพื้นที่ที่ถูกสุ่มออกไปจาก .remaining_areas
		deletearray .remaining_areas[.@idx], 1;
		disablenpc "Effect_NPC#" + .@area_idx;

		// ประกาศผลทอยเต๋า
		switch(rand(1,5)) {
		case 1:
			mapannounce "quiz_02", "@dice : The room's number #" + .@area_idx + " is kicked!", bc_map, 0xEA0000, FW_BOLD, 15, 0, 0;
			break;
		case 2:
			mapannounce "quiz_02", "@dice : Bye bye, #" + .@area_idx + ". See ya!", bc_map, 0xEA0000, FW_BOLD, 15, 0, 0;
			break;
		case 3:
			mapannounce "quiz_02", "@dice : Ohh! #" + .@area_idx + ". Aiming low as usual, I see.", bc_map, 0xEA0000, FW_BOLD, 15, 0, 0;
			break;
		case 4:
			mapannounce "quiz_02", "@dice : Poor guys #" + .@area_idx + ". Sent back to their country!", bc_map, 0xEA0000, FW_BOLD, 15, 0, 0;
			break;
		case 5:
			mapannounce "quiz_02", "@dice : You can only go this far. #" + .@area_idx + " Goodbye!", bc_map, 0xEA0000, FW_BOLD, 15, 0, 0;
			break;
		}
		sleep 1000;

		// ค้นหาผู้เล่นในพื้นที่นั้นและวาร์ปออกไป
		if ((.@area_idx) == 1) {
			areawarp "quiz_02", 289, 263, 300, 256, "sec_in01", 168, 173;
		}
		if ((.@area_idx) == 2) {
			areawarp "quiz_02", 289, 254, 300, 247, "sec_in01", 168, 173;
		}
		if ((.@area_idx) == 3) {
			areawarp "quiz_02", 289, 245, 300, 236, "sec_in01", 168, 173;
		}
		if ((.@area_idx) == 4) {
			areawarp "quiz_02", 307, 263, 318, 256, "sec_in01", 168, 173;
		}
		if ((.@area_idx) == 5) {
			areawarp "quiz_02", 307, 254, 318, 247, "sec_in01", 168, 173;
		}
		if ((.@area_idx) == 6) {
			areawarp "quiz_02", 307, 245, 318, 238, "sec_in01", 168, 173;
		}

		.@rounds++;
		if (.@rounds <= 4) {
			sleep 30000;
			mapannounce "quiz_02", "@dice : Prepare for next dice in 10 seconds.", bc_map, 0xEA0000, FW_BOLD, 15, 0, 0;
		}
	}

	mapannounce "quiz_02", "@dice : How strong... Survivors, claim your rewards now!", bc_map, 0xEA0000, FW_BOLD, 15, 0, 0;
	donpcevent "Dice_System::OnFenceOff";
	hideoffnpc "Dice Manager";
	sleep 5000;
	mapannounce "quiz_02", "@dice : Event closed in 1 minutes, Claim your reward, quickly!", bc_map, 0xEA0000, FW_BOLD, 15, 0, 0;
	mapannounce "sec_in01", "@dice : Event closed in 1 minutes, Claim your reward, quickly!", bc_map, 0xEA0000, FW_BOLD, 15, 0, 0;
	sleep 30000;
	mapannounce "quiz_02", "@dice : Event closed in 30 seconds, Claim your reward, quickly!", bc_map, 0xEA0000, FW_BOLD, 15, 0, 0;
	mapannounce "sec_in01", "@dice : Event closed in 30 seconds, Claim your reward, quickly!", bc_map, 0xEA0000, FW_BOLD, 15, 0, 0;
	sleep 30000;
	
	// ทำการรีเซ็ตคืนค่า Event
	mapwarp "quiz_02", "izlude", 128, 130;
	mapwarp "sec_in01", "izlude", 128, 130;
	hideonnpc "Reward for Dice";
	hideonnpc "Dice Manager";
	deletearray .remaining_areas[0], getarraysize(.remaining_areas);

	.DiceEvent = 0;
	end;
}

izlude,152,136,4	script	Dice Event	874,{
	cutin "zonda_01",2;
	if (getgmlevel() < 98) {
		goto L_Player_View;
	}

	mes "[Dice Event]";
	mes " ";
	mes "สวัสดี GM ต้องการให้ข้าทำอะไร?";
	switch(select("Start Event:Set prize^E50000 ["+getitemname($diceprize)+"]^000000 ^E50000 ["+ $diceprizecount + "]^000000:Test Player View:Nothing")) {

	case 1:
		mes "[Dice Event]";
		mes " ";
		mes "พิมพ์ '^FF0000START^000000' หากต้องการเริ่ม Event";
		input .@confirm$;          
		if (.@confirm$ == "START") {
			donpcevent "Dice_System::OnStart"; 
			close2;
			cutin "",255;
			end;
		}
		mes "[Dice Event]";
		mes " ";
		mes "โอเค ยังก่อนนะ";
		close2;
		cutin "",255;
		end;
		break;
	case 2:
		mes "[Dice Event]";
		mes " ";
		mes "Set ของรางวัลเพื่อแจกคนชนะ";
		mes "กรุณาใส่ ItemID ที่ต้องการ"; 
		input .@guessprizeid; 
		if (getitemname(.@guessprizeid) == "null" || getitemname(.@guessprizeid) == "") {
			mes "[Dice Event]";
			mes " ";
			mes "Error";
			mes "กรุณาใส่ ItemID ที่ต้องการอีกรอบ";
			close;
		}
		set $diceprize, .@guessprizeid;
		mes "[Dice Event]";
		mes " ";
		mes "กรุณาใส่จำนวนที่ต้องการแจก";
		input .@guessprizeamount;
		if (!.@guessprizeamount || .@guessprizeamount == 0 || .@guessprizeamount < 1) {
			set $diceprizecount,1;
		}
		set $diceprizecount,.@guessprizeamount;
		mes "[Dice Event]";
		mes " ";
		mes "The final reward is ...";
		mes "^E50000"+ $diceprizecount + "x " + getitemname($diceprize) +"^000000";
		close2;
		cutin "",255;
		end;
	case 3:
		goto L_Player_View;
	case 4:
		mes "[Dice Event]";
		mes " ";
		mes "Bye!"; 
		close2;
		cutin "",255;
		end;
	}

L_Player_View:
	mes "[Dice Event]";
	mes " ";
	mes "^AA0000No event yet.^000000";
	mes " ";
	mes "Follow up the event at Discord.";
	close2;
	cutin "",255;
	end;

OnInit:
	initnpctimer;

OnTimer1000:
    showscript(" !!! DICE EVENT !!! ");
    setnpctimer 0;
    end;
}
