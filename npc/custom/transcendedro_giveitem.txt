-	script	#GiveItem	-1,{
OnInit:
	bindatcmd "giveitem",strnpcinfo(3)+"::OnGiveItem",99,99;
	end;

OnGiveItem:
	.@gm_id = getcharid(3);
	switch(select(
	
	"Transcended Coin [9955500]",
	"Blacksmith Blessing [6635]",
	"Etel Dust [1000322]",
	"Etel Stone [1000323]",
	"Amethys Fragment [1000405]",
	"Brilliant Snow Flower Magic Stone Extract [1001036]",
	"Cancle"
	
	)) {
		case 1:
			$@item = 9955500;
			break;
		case 2:
			$@item = 6635;
			break;
		case 3:
			$@item = 1000322;
			break;
		case 4:
			$@item = 1000323;
			break;
		case 5:
			$@item = 1000405;
			break;
		case 6:
			$@item = 1001036;
			break;
		case 7:
			mes "Bye!";
			end;
	}
	mes "Input Amount.";
	mes "1 - 5000 ea.";
	input $@amount;
	if (.@item_amount <= 0) {
		.@item_amount = 1;
	} else if (.@item_amount > 5000) {
		.@item_amount = 5000;
	}
	$@total_received = 0;
	addrid(0);
	getitem $@item,$@amount;
	$@total_received++;
	message strcharinfo(0),"[System]: You received ["+ $@amount + "x "+ getitemname($@item) +"] from GM.";
	if (getcharid(3) == .@gm_id) {
		sleep2 100; 
		dispbottom "^FFFF00[Report]: Successfully gave item(s) to [" + $@total_received + "] character(s).^000000";
	}
	end;
}
