Decomposing:
	mes "If you have biolab old weapons like this;";
	mes "^i[1284] ^i[1290] ^i[1285] ^i[1291]";
	mes "you can decomposing it to ^i[6820]";
	deletearray @sold_nameid[0], getarraysize(@sold_nameid);
	close2;
	callshop "Decomposing", 2;
	npcshopattach "Decomposing";
	end;

OnSellItem:
	mes "Items to decomposed:";
	mes "==================";

	deletearray .@valid_items[0], 30000;
	deletearray .@valid_amounts[0], 30000;
	set .@total_points, 0;
	set .@valid_count, 0;
	
	for (set .@i, 0; .@i < getarraysize(@sold_nameid); set .@i, .@i + 1) {
		set .@item_id, @sold_nameid[.@i];
		set .@is_recyclable, 0;
		
		for (set .@j, 0; .@j < getarraysize($decomposable) && !.@is_recyclable; set .@j, .@j + 1) {
			if ($decomposable[.@j] == .@item_id) set .@is_recyclable, 1;
		}
		
		if (.@is_recyclable == 0) {
			mes "  ^FF0000" + @sold_quantity[.@i] + "x " + getitemname(.@item_id) + " (Cannot decomposed.)^000000";
			close;
		} else {
			set .@valid_items[.@valid_count], .@item_id;
			set .@valid_amounts[.@valid_count], @sold_quantity[.@i];
			set .@valid_count, .@valid_count + 1;
			set .@total_points, .@total_points + @sold_quantity[.@i];
			mes "  ^00AA00" + @sold_quantity[.@i] + "x " + getitemname(.@item_id) + "^000000";
		}
	}
	
	deletearray @sold_nameid[0], getarraysize(@sold_nameid);
	deletearray @sold_quantity[0], getarraysize(@sold_quantity);
	
	if (.@valid_count == 0) {
		mes "==================";
		mes "^FF0000No valid items to decomposing!^000000";
		mes "Only Biolab Old Weapons can be decomposed.";
		close;
	}
	
	mes "==================";
	mes "Total: ^0066FF" + .@total_points + " Energy Debris.^000000 ^i[6820]";
	mes "Proceed with decomposing?";
	next;
	
	if (select(" ~ ^AA0000Decomposed it Now!^000000: ~ ^777777Cancel^000000") == 2) {
		mes "What the ......";
		close;
	}

	for (set .@i, 0; .@i < .@valid_count; set .@i, .@i + 1) {
		delitem .@valid_items[.@i], .@valid_amounts[.@i];
	}
	
	getitem 6820, .@total_points;		// Energy Fragment
	
	mes "^00AA00Decomposing successful!^000000";
	mes "     ";
	mes "You gained ^0066FF" + .@total_points + " Energy Debris.^000000 ^i[6820]";
	close;

	initnpctimer;
	setarray $decomposable,
