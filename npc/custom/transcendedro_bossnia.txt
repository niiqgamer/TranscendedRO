izlude,109,140,6	script	Bossnia	908,{
	mes "[ ^0065DFBossnia^000000 ]";
	mes "Which floor would you like to enter?";
	mes "  ";
	mes "<font size=9>^FF0000* You can buy a ticket at Cash Shop^000000</font>";
	next;
	
	switch(select("~ Bossnia F1:~ Bossnia F2:~ Bossnia F3:~ Bossnia F4:~ Exchange Shop:~ Random Card with Flower Left by Phantom:Exit")) {
		case 1: set .@floor,1; break;
		case 2: set .@floor,2; break;
		case 3: set .@floor,3; break;
		case 4: set .@floor,4; break;
		case 5:
			close2;
			callshop "bossnia";
			end;
		case 6:
			mes "How many times to random?";
			mes "(1 - 500)";
			next;
			input .@rolls;
			callfunc("RandomCardBossnia", .@rolls);
		case 7:	close;
	}
	
	getinventorylist;
	set .@idx, .@floor - 1;
	set .@price, $@price[.@idx];

	if (rentalcountitem($@price[.@idx]) < 1) {
		mes "You don't have any ticket";
		mes "or your ticket has expired.";
		mes "You need at least 1x ^i[" + $@price[.@idx] + "] to enter.";
		close;
	}
	
	mes "Ticket verified.";
	mes "^00AA00P a s s !^000000";
	mes "Warping you to Floor " + .@floor + " ...";
	close2;
	warp $@floormap$[.@idx], 0, 0;
	end;

OnInit:
	initnpctimer;
	end;

OnTimer1000:
    showscript(" ! BOSSNIA ! ");
    setnpctimer 0;
    end;
}

		}

	}

	return 1;

// Manager script: spawns & keeps MVPs alive on all 4 floors.
// ========================================================
-	script	BossNia_Manager	-1,{
	end;

OnInit:
	// --- CONFIGURATION ---
	setarray $@floormap$[0], "bossnia_01", "bossnia_02", "bossnia_03", "bossnia_04";
	setarray $@price[0], 9900011, 9900012, 9900013, 9900014;
	setarray $@mvp_count[0], 30, 45, 60, 75;
	setarray $@relieve_lv[0], 2, 4, 6, 10;

	setarray $@mobs_f1[0], 1252,1768,1272,1046,1389;
	setarray $@mobs_f2[0], 1252,1768,1272,1046,1389,1112,1719,1147,1059,1039,1658,1511,1785,1685,1115;
	setarray $@mobs_f3[0], 20943,20536,3241,3245,3246,3221,3223,3224,3240,3242,3243,3244,3220,3222,3225;
	setarray $@mobs_f4[0], 20621,20422,20572,20610,20611,21531,21533,20934,20573,21981,21996;
	// ---------------------

	callsub S_InitAllFloors;
	end;

S_InitAllFloors:
	callsub S_SpawnFloor, 1;
	callsub S_SpawnFloor, 2;
	callsub S_SpawnFloor, 3;
	callsub S_SpawnFloor, 4;
	return;

S_SpawnFloor:
	set .@floor, getarg(0);
	set .@idx, .@floor - 1;
	set .@count, $@mvp_count[.@idx];

	if (.@floor == 1) set .@label$, "BossNia_Manager::OnMobDeadF1";
	else if (.@floor == 2) set .@label$, "BossNia_Manager::OnMobDeadF2";
	else if (.@floor == 3) set .@label$, "BossNia_Manager::OnMobDeadF3";
	else set .@label$, "BossNia_Manager::OnMobDeadF4";

	for ( .@i = 0; .@i < .@count; .@i++ ) {
		if (.@floor == 1) {
			set .@mobsz, getarraysize($@mobs_f1);
			set .@mobid, $@mobs_f1[ rand(0, .@mobsz-1) ];
		}
		else if (.@floor == 2) {
			set .@mobsz, getarraysize($@mobs_f2);
			set .@mobid, $@mobs_f2[ rand(0, .@mobsz-1) ];
		}
		else if (.@floor == 3) {
			set .@mobsz, getarraysize($@mobs_f3);
			set .@mobid, $@mobs_f3[ rand(0, .@mobsz-1) ];
		}
		else {
			set .@mobsz, getarraysize($@mobs_f4);
			set .@mobid, $@mobs_f4[ rand(0, .@mobsz-1) ];
		}

		monster $@floormap$[.@idx], 0, 0, "--ja--", .@mobid, 1, .@label$;
		unitskilluseid $@mobid[0], "NPC_RELIEVE_ON", $@relieve_lv[.@idx];
	}
	return;

// ========================================================
// OnMobDead handlers per floor
// ========================================================
OnMobDeadF1:
	if (playerattached()) {
		getitem 25503,1;
	}
	set .@newmobid, $@mobs_f1[ rand(0, getarraysize($@mobs_f1)-1) ];
	monster "bossnia_01", 0, 0, "--ja--", .@newmobid, 1, "BossNia_Manager::OnMobDeadF1";
	unitskilluseid $@mobid[0], "NPC_RELIEVE_ON", $@relieve_lv[0];
	end;

OnMobDeadF2:
	if (playerattached()) {
		getitem 25503,rand(2,3);
	}
	set .@newmobid, $@mobs_f2[ rand(0, getarraysize($@mobs_f2)-1) ];
	monster "bossnia_02", 0, 0, "--ja--", .@newmobid, 1, "BossNia_Manager::OnMobDeadF2";
	unitskilluseid $@mobid[0], "NPC_RELIEVE_ON", $@relieve_lv[1];
	end;

OnMobDeadF3:
	if (playerattached()) {
		if (rand(0,2) == 0) {
			getitem 1001652,1;
		} else {
			getitem 25503,rand(2,4);
		}
	}
	set .@newmobid, $@mobs_f3[ rand(0, getarraysize($@mobs_f3)-1) ];
	monster "bossnia_03", 0, 0, "--ja--", .@newmobid, 1, "BossNia_Manager::OnMobDeadF3";
	unitskilluseid $@mobid[0], "NPC_RELIEVE_ON", $@relieve_lv[2];
	end;

OnMobDeadF4:
	if (playerattached()) {
		if (rand(0,2) == 0) {
			getitem 1001652,2;
		} else {
			getitem 25503,rand(3,5);
		}
	}
	set .@newmobid, $@mobs_f4[ rand(0, getarraysize($@mobs_f4)-1) ];
	monster "bossnia_04", 0, 0, "--ja--", .@newmobid, 1, "BossNia_Manager::OnMobDeadF4";
	unitskilluseid $@mobid[0], "NPC_RELIEVE_ON", $@relieve_lv[3];
	end;
}

-	script	BossNia_TicketGuard	-1,{
OnPCLoadMapEvent:
	// หาว่าแมพปัจจุบันตรงกับ $@floormap$ ไหน
	.@mapname$ = strcharinfo(3);
	.@idx = -1;
	for (.@i = 0; .@i < 4; .@i++) {
		if ($@floormap$[.@i] == .@mapname$) {
			.@idx = .@i;
			break;
		}
	}
	if (.@idx < 0) end; // ไม่ใช่แมพ bossnia

	.@ticket = $@price[.@idx];

	// เช็คตั๋วใน inventory
	getinventorylist;
	.@permanent = 0;
	.@expire = 0;
	for (.@i = 0; .@i < @inventorylist_count; .@i++) {
		if (@inventorylist_id[.@i] != .@ticket) continue;
		if (@inventorylist_expire[.@i] == 0) {
			.@permanent = 1;
			break;
		} else if (@inventorylist_expire[.@i] > gettimetick(2)) {
			if (@inventorylist_expire[.@i] > .@expire)
				.@expire = @inventorylist_expire[.@i];
		}
	}

	// ลบ timer เก่าก่อน (ถ้ามี)
	deltimer "BossNia_TicketGuard::OnTicketExpired";

	// กรณี 1: ไม่มีตั๋วเลย → เตะออกทันที
	if (!.@permanent && .@expire == 0) {
		dispbottom "[ BOSSNIA ] You don't have a valid ticket. Warping you out.";
		warp "SavePoint", 0, 0;
		end;
	}

	// กรณี 2: มีตั๋วถาวร → ไม่เตะ
	if (.@permanent) {
		end;
	}

	// กรณี 3: มีตั๋วเช่า → คำนวณเวลาเหลือแล้วตั้ง timer
	.@left = .@expire - gettimetick(2);
	if (.@left <= 0) {
		dispbottom "[ BOSSNIA ] Your ticket has expired. Warping you out.";
		warp "SavePoint", 0, 0;
		end;
	}

	// ตั้ง timer ให้เตะออกตอนหมดอายุพอดี (convert วินาที → มิลลิวินาที)
	addtimer (.@left * 1000), "BossNia_TicketGuard::OnTicketExpired";
	dispbottom "[ BOSSNIA ] Your ticket will expire in " + .@left + " seconds.";
	end;

OnTicketExpired:
	// ตรวจสอบว่ายังอยู่ในแมพ bossnia หรือไม่
	.@mapname$ = strcharinfo(3);
	.@idx = -1;
	for (.@i = 0; .@i < 4; .@i++) {
		if ($@floormap$[.@i] == .@mapname$) {
			.@idx = .@i;
			break;
		}
	}
	if (.@idx < 0) end; // ออกแมพไปแล้ว ไม่ต้องทำอะไร

	// ตรวจสอบตั๋วอีกครั้ง (เผื่อได้ใหม่ในระหว่างนั้น)
	.@ticket = $@price[.@idx];
	if (countitem(.@ticket) >= 1) {
		dispbottom "[ BOSSNIA ] Ticket renewed. Timer reset.";
		end;
	}

	dispbottom "[ BOSSNIA ] Your ticket has expired!";
	warp "SavePoint", 0, 0;
	end;
}
